    <script>
        let players = [];

        function addTeam() {
            const teamName = prompt("Nombre del equipo rival:");
            if (teamName) {
                const div = document.createElement('div');
                div.className = "p-3 text-gray-300 hover:bg-gray-800 cursor-pointer rounded-md flex items-center gap-2";
                div.innerHTML = "🏀 " + teamName;
                document.getElementById('team-list').appendChild(div);
            }
        }

        function openPlayerModal() { document.getElementById('modal').classList.remove('hidden'); }
        function closePlayerModal() { document.getElementById('modal').classList.add('hidden'); }

        function savePlayer() {
            const player = {
                id: Date.now(),
                name: document.getElementById('in-name').value.toUpperCase(),
                num: document.getElementById('in-num').value,
                pos: document.getElementById('in-pos').value,
                str: document.getElementById('in-str').value,
                weak: document.getElementById('in-weak').value
            };
            if(player.name) {
                players.push(player);
                renderPlayers();
                closePlayerModal();
                document.querySelectorAll('#modal input, #modal textarea').forEach(i => i.value = '');
            }
        }

        function renderPlayers() {
            const list = document.getElementById('player-list');
            list.innerHTML = players.map(p => `
                <div onclick="viewPlayer(${p.id})" class="p-4 bg-white border border-gray-200 rounded-xl hover:border-orange-500 hover:shadow-md cursor-pointer transition flex items-center gap-4">
                    <span class="text-xl font-black text-orange-500">#${p.num}</span>
                    <span class="font-bold text-gray-800">${p.name}</span>
                </div>
            `).join('');
        }

        function viewPlayer(id) {
            const p = players.find(x => x.id === id);
            document.getElementById('welcome-screen').classList.add('hidden');
            document.getElementById('scouting-content').classList.remove('hidden');
            document.getElementById('p-display-name').innerText = p.name;
            document.getElementById('p-display-pos').innerText = p.pos;
            document.getElementById('p-display-str').innerText = p.str;
            document.getElementById('p-display-weak').innerText = p.weak;
        }
    </script>
